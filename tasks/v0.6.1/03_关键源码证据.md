 122:                        for c in conds])
 123:         ha = np.stack([self.by_cond[c][1][rng.integers(0, len(self.by_cond[c][1]))]
 124:                        for c in conds])

  38: def derive(path):
  39:     d = np.load(path)
  40:     t = d["timestamp"] - d["timestamp"][0]
  41:     dt = float(np.median(np.diff(t)))
  42:     dur = t[-1] + dt

  44:     # previous-sample hold 索引（严格因果）

  46: 
  47:     out = {"timestamp": grid.astype(np.float64),
  48:            "episode_time": grid.astype(np.float64),
  49:            "phase": d["phase"][idx_hold],

  52:         out[f"in_{k}"] = causal_lowpass(d[k].astype(np.float64), dt, FC)[idx_hold].astype(np.float32)
  53:     # labels 版对齐到 inputs 网格：resample_poly 输出 j 对应 t=j*0.05（与 grid 同原点），
  54:     # 长度按 ceil(n*2/5)，与 grid 最多差 1 个尾样本 -> 截断或末端 edge-hold 补齐

  56:     for k in LABEL_KEYS:

  60:         else:

  62:         out[f"lb_{k}"] = lb.astype(np.float32)
  63:     # execution/residual 同时给因果版（训练标签一致性可选）
  64:     for k in ("execution", "residual"):
  65:         out[f"in_{k}"] = causal_lowpass(d[k].astype(np.float64), dt, FC)[idx_hold].astype(np.float32)

  67:         if k.startswith("anchor_"):
  68:             out[k] = d[k]
  69:     return out, len(grid), T20

  53: def predict_with(model_name, model, aux, hp, ha, fa, fric, c=None):
  54:     with torch.no_grad():
  55:         if model_name == "M0":
  56:             r = model(hp, ha, fa)["r_hat"]
  57:         else:
  58:             if c is None:
  59:                 c = model.encode_context(hp, ha)
  60:             if model_name == "M3":
  61:                 c = c + aux(fric.reshape(-1, 1))
  62:             z = model.encode_state(hp)
  63:             r = model.predict_execution(z, c, fa)["r_hat"]
  64:     return fa + r
  65: 
  66: 
  67: def fit_arx(train_data, device):
  68:     """ridge: [last proprio(40), future cmd flat(120)] -> residual flat(120)，train-only。"""
  69:     n = len(train_data)
  70:     idx = np.arange(n)
  71:     hp, ha, fa, fr, _, _, _, _ = train_data.batch(idx, "cpu")
  72:     X = np.concatenate([hp[:, -1, :].numpy(), fa.reshape(n, -1).numpy()], axis=1)
  73:     Y = fr.reshape(n, -1).numpy()
  74:     Xb = np.concatenate([X, np.ones((n, 1))], axis=1)
  75:     lam = 1.0
  76:     W = np.linalg.solve(Xb.T @ Xb + lam * np.eye(Xb.shape[1]), Xb.T @ Y)
  77:     return W
  78: 
  79: 
  80: def arx_predict(W, hp, fa):
  81:     n = len(hp)
  82:     X = np.concatenate([hp[:, -1, :].cpu().numpy(), fa.reshape(n, -1).cpu().numpy()], axis=1)
  83:     Xb = np.concatenate([X, np.ones((n, 1))], axis=1)
  84:     return fa.cpu().numpy() + (Xb @ W).reshape(n, H, 3)
  85: 
  86: 
  87: def mae_rows(pred, gt, names):
  88:     err = np.abs(pred - gt)
  89:     out = {}
  90:     for lname, k in LEADS.items():
  91:         for i, n in enumerate(VEL):
  92:             out[f"MAE_{n}@{lname}"] = float(err[:, :k, i].mean())
  93:     out["MAE_all@2s"] = float(err.mean())
  94:     return out

 198: def run_wave(env, unwrapped, policy, cmd_term, segs, settle_s, control_dt,
 199:              foot_ids, sensor_foot_ids):
 200:     """执行一条 settle+schedule，返回每 env 的 buffer/term_reason。"""
 201:     n_steps = int(math.ceil((settle_s + sum(d for d, _ in segs)) / control_dt))
 202:     buffers = [{k: [] for k in BUFFER_KEYS} for _ in range(unwrapped.num_envs)]
 203:     alive = np.ones(unwrapped.num_envs, dtype=bool)
 204:     term = ["schedule_end"] * unwrapped.num_envs
 205:     obs = env.get_observations()
 206:     t_abs = 0.0
 207:     for step in range(n_steps):
 208:         t = step * control_dt
 209:         if t < settle_s:
 210:             cmd_np = np.zeros((unwrapped.num_envs, 3), dtype=np.float32)
 211:             phase = 0
 212:         else:
 213:             cmd_np = np.stack([cmd_at(segs, t - settle_s)] * unwrapped.num_envs)
 214:             phase = 1
 215:         cmd_term.vel_command_b = torch.as_tensor(cmd_np, device=unwrapped.device)
 216:         rec = capture(unwrapped, cmd_np, phase, t_abs, t, foot_ids, sensor_foot_ids)
 217:         with torch.inference_mode():
 218:             actions = policy(obs)
 219:         obs, _, dones, _ = env.step(actions)
 220:         t_abs += control_dt
 221:         for e in range(unwrapped.num_envs):
 222:             if not alive[e]:
 223:                 continue
 224:             for k in BUFFER_KEYS:
 225:                 v = rec[k][e]
 226:                 buffers[e][k].append(v.copy() if isinstance(v, np.ndarray) else v)
 227:             if bool(dones[e]):
 228:                 alive[e] = False
 229:                 term[e] = "terminated"
